function bytesToBits(bytes){ let bits = ''; for(const byte of bytes){ bits += byte .toString(2) .padStart(8,'0'); } return bits; } function encode(){ const file = document.getElementById( 'encodeImage' ).files[0]; const message = document.getElementById( 'message' ).value; if(!file || !message){ alert( 'Select image and enter message' ); return; } const encoder = new TextEncoder(); const messageBytes = encoder.encode(message); const length = messageBytes.length; const payload = new Uint8Array( length + 4 ); payload[0] = (length >> 24) & 255; payload[1] = (length >> 16) & 255; payload[2] = (length >> 8) & 255; payload[3] = length & 255; payload.set( messageBytes, 4 ); const bits = bytesToBits(payload); const img = new Image(); img.onload = () => { const canvas = document.getElementById( 'canvas' ); const ctx = canvas.getContext( '2d' ); canvas.width = img.width; canvas.height = img.height; ctx.drawImage( img, 0, 0 ); const imageData = ctx.getImageData( 0, 0, canvas.width, canvas.height ); const data = imageData.data; if( bits.length > data.length / 4 ){ alert( 'Message too large for image' ); return; } for( let i = 0; i < bits.length; i++ ){ data[i * 4] = ( data[i * 4] & 254 ) | Number(bits[i]); } ctx.putImageData( imageData, 0, 0 ); canvas.toBlob(blob => { const url = URL.createObjectURL( blob ); const preview = document.getElementById( 'preview' ); preview.src = url; preview.style.display = 'block'; const btn = document.getElementById( 'downloadBtn' ); btn.href = url; btn.download = 'ghost-message.png'; btn.style.display = 'block'; },'image/png'); }; img.src = URL.createObjectURL( file ); } function decode(){ const file = document.getElementById( 'decodeImage' ).files[0]; if(!file){ alert( 'Select image' ); return; } const img = new Image(); img.onload = () => { const canvas = document.getElementById( 'canvas' ); const ctx = canvas.getContext( '2d' ); canvas.width = img.width; canvas.height = img.height; ctx.drawImage( img, 0, 0 ); const data = ctx.getImageData( 0, 0, canvas.width, canvas.height ).data; let bitStream = ''; for( let i = 0; i < data.length; i += 4 ){ bitStream += (data[i] & 1); } const lengthBytes = []; for( let i = 0; i < 32; i += 8 ){ lengthBytes.push( parseInt( bitStream.slice( i, i + 8 ), 2 ) ); } const messageLength = ( lengthBytes[0] << 24 ) | ( lengthBytes[1] << 16 ) | ( lengthBytes[2] << 8 ) | ( lengthBytes[3] ); const bytes = new Uint8Array( messageLength ); let pointer = 32; for( let i = 0; i < messageLength; i++ ){ bytes[i] = parseInt( bitStream.slice( pointer, pointer + 8 ), 2 ); pointer += 8; } const decoder = new TextDecoder(); const message = decoder.decode( bytes ); document.getElementById( 'decoded' ).innerText = message || 'No hidden message found'; }; img.src = URL.createObjectURL( file ); }
